# 放在 nginx 的 http {} 内；替换域名和证书路径后验证。
limit_req_zone $binary_remote_addr zone=ops_login:10m rate=5r/m;
limit_conn_zone $binary_remote_addr zone=ops_conn:10m;

server {
    listen 443 ssl;
    server_name ops.example.test;
    if ($host != ops.example.test) { return 444; }
    ssl_certificate /etc/ops-course/tls/fullchain.pem;
    ssl_certificate_key /etc/ops-course/tls/privkey.pem;
    ssl_protocols TLSv1.2 TLSv1.3;
    client_max_body_size 8k;
    client_header_timeout 10s;
    client_body_timeout 10s;
    limit_conn ops_conn 20;
    limit_req_status 429;

    location = /api/login {
        limit_req zone=ops_login burst=3 nodelay;
        proxy_pass http://127.0.0.1:8096;
        proxy_set_header Host ops.example.test;
        proxy_set_header X-Forwarded-Proto https;
        proxy_read_timeout 10s;
        proxy_connect_timeout 2s;
        proxy_next_upstream off;
    }
    location / {
        proxy_pass http://127.0.0.1:8096;
        proxy_set_header Host ops.example.test;
        proxy_set_header X-Forwarded-Proto https;
        proxy_read_timeout 10s;
        proxy_connect_timeout 2s;
        proxy_next_upstream off;
    }
}
